# 의존성 설치 (최초 1회)
# 가상환경은 backend/.venv 에 둔다 (.vscode/settings.json의 python.defaultInterpreterPath가
# 이 경로를 가리키므로 VSCode가 자동으로 인터프리터를 잡는다).

python -m venv backend/.venv
backend\.venv\Scripts\pip install -r requirements.txt

# 이후 실행 (항상 backend/.venv의 python 사용)

# 백엔드 (프로젝트 루트에서)
backend\.venv\Scripts\python -m backend.server

# 프론트엔드 (frontend/ 폴더에서)
npm run dev

# 참고: 이 개발 PC에는 C:\WEVE\...\Config.ini(실장비 설정)가 없어서 backend.server
# 전체 임포트(하드웨어 매니저 경유)는 NoSectionError로 실패한다 — 실장비가 있는
# 실험실 PC에서만 정상 기동한다. 멀티에이전트 파이프라인(backend.agents.*)은
# 하드웨어 설정과 무관하게 독립적으로 임포트/실행 가능하다(에이전트 개발/테스트는
# 이 경로로 진행).
